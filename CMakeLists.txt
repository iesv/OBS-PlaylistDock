cmake_minimum_required(VERSION 3.24)
project(OBS-PlaylistDock VERSION 1.2.0 LANGUAGES CXX)

if((NOT WIN32 AND NOT APPLE) OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This release requires Windows x64 or macOS ARM64.")
endif()
if(APPLE)
  if(NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES arm64)
  elseif(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "macOS builds require arm64.")
  endif()
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 6.8.3 EXACT CONFIG REQUIRED COMPONENTS Core Widgets)

add_library(playlistdock-ui STATIC src/PlaylistDock.cpp src/PlaylistDock.hpp src/RevealFile.cpp src/RevealFile.hpp)
set_target_properties(playlistdock-ui PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_link_libraries(playlistdock-ui PUBLIC OBS::libobs Qt6::Core Qt6::Widgets)
if(WIN32)
  target_link_libraries(playlistdock-ui PRIVATE shell32 ole32)
endif()
target_include_directories(playlistdock-ui PUBLIC src)
target_compile_definitions(playlistdock-ui PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII PLAYLISTDOCK_VERSION="${PROJECT_VERSION}")

add_library(OBS-PlaylistDock MODULE src/plugin-main.cpp)
target_link_libraries(OBS-PlaylistDock PRIVATE playlistdock-ui OBS::obs-frontend-api)
target_compile_definitions(OBS-PlaylistDock PRIVATE PLAYLISTDOCK_VERSION="${PROJECT_VERSION}")
set_target_properties(OBS-PlaylistDock PROPERTIES PREFIX "")

if(MSVC)
  target_compile_options(playlistdock-ui PRIVATE /utf-8 /W4 /permissive-)
  target_compile_options(OBS-PlaylistDock PRIVATE /utf-8 /W4 /permissive-)
else()
  target_compile_options(playlistdock-ui PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(OBS-PlaylistDock PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(APPLE)
  set_target_properties(OBS-PlaylistDock PROPERTIES
    BUNDLE TRUE BUNDLE_EXTENSION plugin OUTPUT_NAME obs-playlistdock
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/data/Info.plist.in"
    INSTALL_RPATH "@executable_path/../Frameworks")
  set(PLAYLISTDOCK_PLATFORM macos-arm64)
  set(PLAYLISTDOCK_DATA obs-playlistdock.plugin/Contents/Resources)
  configure_file(data/Info.plist.in ${CMAKE_CURRENT_BINARY_DIR}/Info.plist @ONLY)
  install(TARGETS OBS-PlaylistDock BUNDLE DESTINATION . LIBRARY DESTINATION .)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/Info.plist DESTINATION obs-playlistdock.plugin/Contents)
else()
  configure_file(data/version.rc.in ${CMAKE_CURRENT_BINARY_DIR}/version.rc @ONLY)
  target_sources(OBS-PlaylistDock PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/version.rc)
  set(PLAYLISTDOCK_PLATFORM windows-x64)
  set(PLAYLISTDOCK_DATA obs-playlistdock/data)
  install(TARGETS OBS-PlaylistDock RUNTIME DESTINATION obs-playlistdock/bin/64bit LIBRARY DESTINATION obs-playlistdock/bin/64bit)
endif()
configure_file(data/manifest.json.in ${CMAKE_CURRENT_BINARY_DIR}/manifest.json @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/manifest.json DESTINATION ${PLAYLISTDOCK_DATA})
install(DIRECTORY data/locale DESTINATION ${PLAYLISTDOCK_DATA})
install(FILES README.md LICENSE DESTINATION .)

option(PLAYLISTDOCK_BUILD_TESTS "Build stress and integration tests" OFF)
if(PLAYLISTDOCK_BUILD_TESTS)
  enable_testing()
  add_executable(playlistdock-stress tests/stress.cpp)
  set_target_properties(playlistdock-stress PROPERTIES AUTOMOC ON)
  target_link_libraries(playlistdock-stress PRIVATE playlistdock-ui)
  if(WIN32)
    target_link_libraries(playlistdock-stress PRIVATE psapi ole32)
  endif()
  target_compile_definitions(playlistdock-stress PRIVATE PLAYLISTDOCK_TEST_LOCALE="${CMAKE_CURRENT_SOURCE_DIR}/data/locale/zh-CN.ini")
  target_compile_options(playlistdock-stress PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  add_test(NAME playlistdock-stress COMMAND playlistdock-stress --fake)
  add_library(playlistdock-smoke-exit MODULE tests/smoke-exit.cpp)
  set_target_properties(playlistdock-smoke-exit PROPERTIES PREFIX "")
  target_link_libraries(playlistdock-smoke-exit PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets)
  target_compile_options(playlistdock-smoke-exit PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
endif()
